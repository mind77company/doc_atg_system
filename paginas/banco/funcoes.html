Funções do banco (RPCs e apoio)

As funções que a plataforma, o n8n e o próprio banco executam

Resumo simples

Algumas tarefas são feitas por funções guardadas no próprio banco, em vez de pelo navegador. Isso garante que um conjunto de passos aconteça junto: por exemplo, criar um gerador e já registrar no histórico quem o criou.

Há três tipos: funções que a plataforma chama (criar gerador, salvar preditiva, validar cadastro), funções que o n8n chama (controle do limite do iClass) e funções de apoio à importação da planilha de preditivas e à comparação de nomes.

consumir_rate_limit_iclass

Resumo simples

Controla o limite de 25 chamadas ao iClass por minuto: diz se a chamada pode ser feita agora ou quantos segundos esperar.

Detalhes técnicos: assinatura, regras e código
ItemValor
Parâmetroslimite integer DEFAULT 25, janela_segundos integer DEFAULT 60
DevolveTABLE(permitido boolean, espera_segundos numeric)
Linguagemplpgsql
Permissão ao executarSECURITY DEFINER (usa a permissão do dono da função, ignora o RLS)
search_path fixonão definido
VolatilidadeVOLATILE
Quem pode executarPUBLIC, anon, authenticated
Chamada pelo código do frontnão (ou só por fluxos/outras funções)

Regras

  • Valores padrão: limite 25 chamadas e janela de 60 segundos.
  • Um único contador para todos os fluxos.
  • Trava a linha para duas chamadas simultâneas não passarem juntas.
  • Roda com a permissão do dono (SECURITY DEFINER) e não define search_path (diferente das três funções chamadas pelo front).

Código da função

CREATE OR REPLACE FUNCTION public.consumir_rate_limit_iclass(limite integer DEFAULT 25, janela_segundos integer DEFAULT 60)
 RETURNS TABLE(permitido boolean, espera_segundos numeric)
 LANGUAGE plpgsql
 SECURITY DEFINER
AS $function$
declare
  linha rate_limit_iclass%rowtype;
begin
  select * into linha from rate_limit_iclass where id = 1 for update;

  if now() - linha.janela_inicio >= (janela_segundos || ' seconds')::interval then
    update rate_limit_iclass set janela_inicio = now(), contagem = 1 where id = 1;
    return query select true, 0::numeric;
  elsif linha.contagem < limite then
    update rate_limit_iclass set contagem = contagem + 1 where id = 1;
    return query select true, 0::numeric;
  else
    return query select false,
      extract(epoch from (linha.janela_inicio + (janela_segundos || ' seconds')::interval - now()));
  end if;
end;
$function$

CREATE OR REPLACE FUNCTION public.criar_gerador(p_omie_cliente_id bigint, p_ativo_id bigint, p_dados jsonb, p_usuario text)
 RETURNS geradores_atg
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare r public.geradores_atg;
begin
  if p_ativo_id is null then
    raise exception 'ativo_id obrigatorio: gerador so pode ser criado com ativo do iClass';
  end if;

  -- herda cnpj/nome do cliente
  insert into public.geradores_atg (
    omie_cliente_id, ativo_id,
    nome_fantasia, razao_social, cnpj_limpo,
    iclass_id_cliente, iclass_codigo_cliente,
    num_serie, num_patrimonio, descricao,
    fabricante, modelo, status_ativo, tipo_equipamento,
    processado_em, atualizado_em
  )
  select
    o.id, p_ativo_id,
    o.nome_fantasia, o.razao_social, o.cnpj_limpo,
    o.iclass_id_encontrado, o.iclass_codigo_encontrado,
    p_dados->>'num_serie', p_dados->>'num_patrimonio', p_dados->>'descricao',
    p_dados->>'fabricante', p_dados->>'modelo',
    coalesce(p_dados->>'status_ativo','ATIVO'), p_dados->>'tipo_equipamento',
    now(), now()
  from public.omie_clientes o
  where o.id = p_omie_cliente_id
  returning * into r;

  if not found then
    raise exception 'cliente % nao encontrado', p_omie_cliente_id;
  end if;

  insert into public.log_edicoes (entidade, entidade_id, acao, usuario, detalhe)
  values ('gerador', r.id, 'criou_gerador', p_usuario,
          jsonb_build_object('ativo_id', p_ativo_id, 'omie_cliente_id', p_omie_cliente_id));

  return r;
end $function$

match_cliente

Resumo simples

Dado um nome de cliente da planilha, procura os geradores mais parecidos e decide se liga sozinho, pede revisão ou desiste.

Detalhes técnicos: assinatura, regras e código
ItemValor
Parâmetrosp_cliente text
DevolveTABLE(gerador_id bigint, omie_cliente_id bigint, cnpj_limpo text, score numeric, match_status text, candidatos jsonb)
Linguagemplpgsql
Permissão ao executarpermissão de quem chama
search_path fixo'public', 'extensions', 'pg_catalog'
VolatilidadeSTABLE
Quem pode executarPUBLIC, anon, authenticated
Chamada pelo código do frontnão (ou só por fluxos/outras funções)

Regras

  • Filtro rápido: similaridade acima de 0,15 ou palavra em comum com o nome fantasia.
  • Pega os 3 melhores candidatos e decide pelo melhor: nota abaixo de 0,75 = sem_match; nome com sufixo G1/G2 = revisar; nota de 0,90 ou mais e CNPJ com exatamente um gerador = auto; senão revisar.
  • Só no caso auto devolve o gerador_id.

Código da função

CREATE OR REPLACE FUNCTION public.match_cliente(p_cliente text)
 RETURNS TABLE(gerador_id bigint, omie_cliente_id bigint, cnpj_limpo text, score numeric, match_status text, candidatos jsonb)
 LANGUAGE plpgsql
 STABLE
 SET search_path TO 'public', 'extensions', 'pg_catalog'
AS $function$
declare
  v_norm   text   := public.norm_nome(p_cliente);
  v_sufixo text   := public.sufixo_gerador(p_cliente);
  v_tp     text[] := public.tokens_nome(p_cliente);
begin
  return query
  with cand as (
    -- pre-filtro barato para nao rodar score_nomes contra os 458.
    -- trigram com threshold folgado (0.15) + rede de seguranca por token:
    -- nomes curtos no banco ('AFAM') nao passam no trigram contra uma
    -- planilha verbosa, mas compartilham token.
    select g.id, g.omie_cliente_id, g.cnpj_limpo,
           g.nome_fantasia, g.razao_social,
           greatest(
             public.score_nomes(p_cliente, g.nome_fantasia),
             public.score_nomes(p_cliente, g.razao_social)
           ) as sc
    from public.geradores_atg g
    where similarity(public.norm_nome(g.nome_fantasia), v_norm) > 0.15
       or similarity(public.norm_nome(g.razao_social),  v_norm) > 0.15
       or public.tokens_nome(g.nome_fantasia) && v_tp
  ),
  top3 as (
    select * from cand order by sc desc, id asc limit 3
  ),
  best as (
    select t.*,
           (select count(*) from public.geradores_atg x
            where x.cnpj_limpo = t.cnpj_limpo) as n_ger
    from top3 t order by t.sc desc, t.id asc limit 1
  ),
  decidido as (
    select b.*,
      case
        when b.sc < 0.75            then 'sem_match'
        when v_sufixo is not null   then 'revisar'   -- G1/G2: banco nao distingue
        when b.sc >= 0.90 and b.n_ger = 1 then 'auto'
        else 'revisar'
      end as st
    from best b
  ),
  cands_json as (
    select coalesce(jsonb_agg(jsonb_build_object(
             'gerador_id',       t.id,
             'nome_fantasia',    t.nome_fantasia,
             'razao_social',     t.razao_social,
             'cnpj',             t.cnpj_limpo,
             'score',            t.sc,
             'n_geradores_cnpj', (select count(*) from public.geradores_atg x
                                  where x.cnpj_limpo = t.cnpj_limpo))
             order by t.sc desc), '[]'::jsonb) as js
    from top3 t
  )
  select
    case when d.st = 'auto' then d.id              else null::bigint end,
    case when d.st = 'auto' then d.omie_cliente_id else null::bigint end,
    case when d.st = 'sem_match' then null::text   else d.cnpj_limpo end,
    d.sc,
    d.st,
    c.js
  from decidido d cross join cands_json c;

  -- nenhum candidato passou nem no pre-filtro
  if not found then
    return query select null::bigint, null::bigint, null::text,
                        0::numeric, 'sem_match'::text, '[]'::jsonb;
  end if;
end $function$

parse_data_pred

Resumo simples

Converte o texto de data da planilha em data. Textos vazios, "00/00/0000", "REF", "null" e "não aplica" viram vazio.

Detalhes técnicos: assinatura, regras e código

Regras

  • Tenta primeiro a conversão padrão do banco e só depois DD/MM/YYYY. Não confirmado Se o banco interpretar datas como mês/dia, 01/02/2026 seria lido como 2 de janeiro. Conferir com show datestyle.

Código da função

CREATE OR REPLACE FUNCTION public.parse_data_pred(txt text)
 RETURNS date
 LANGUAGE plpgsql
 IMMUTABLE
AS $function$
declare t text := trim(coalesce(txt, ''));
begin
  if t = '' or t ilike '00/00/0000' or t ilike '%aplica%'
     or t ilike '%REF%' or t ilike 'null' then
    return null;
  end if;
  begin return t::date;                 exception when others then null; end;
  begin return to_date(t,'DD/MM/YYYY'); exception when others then null; end;
  begin return to_date(t,'YYYY-MM-DD'); exception when others then null; end;
  return null;
end $function$

Código da função

CREATE OR REPLACE FUNCTION public.processar_preditivas(p_import_id uuid)
 RETURNS TABLE(linhas integer, auto integer, revisar integer, sem_match integer)
 LANGUAGE plpgsql
 SET search_path TO 'public', 'extensions', 'pg_catalog'
AS $function$
begin
  delete from public.preditivas_atg     where import_id = p_import_id;
  delete from public.oxicatalisador_atg where import_id = p_import_id;

  -- match calculado 1x por cliente distinto, reaproveitado nos dois inserts
  create temp table _mt on commit drop as
  select s.cliente, m.*
  from (select distinct cliente from public.preditivas_staging
        where import_id = p_import_id and cliente is not null) s
  cross join lateral public.match_cliente(s.cliente) m;

  -- ---------------- PREDITIVAS (7 itens, sem oxicatalisador) ----------
  with unpiv as (
    select s.*, v.item_codigo, v.dt_real, v.dt_venc, v.obs, v.st
    from public.preditivas_staging s
    cross join lateral (values
      ('oleo_filtros',        s.oleo_realizada,  s.oleo_vencimento,  s.oleo_obs,  s.oleo_status),
      ('bateria',             s.bat_realizada,   s.bat_vencimento,   s.bat_obs,   s.bat_status),
      ('mang_combustivel',    s.mcomb_realizada, s.mcomb_vencimento, s.mcomb_obs, s.mcomb_status),
      ('limpeza_qta',         s.qta_realizada,   s.qta_vencimento,   s.qta_obs,   s.qta_status),
      ('mang_preaquecimento', s.mpre_realizada,  s.mpre_vencimento,  s.mpre_obs,  s.mpre_status),
      ('limpeza_radiador',    s.rad_realizada,   s.rad_vencimento,
         nullif(concat_ws(' | ', nullif(s.rad_obs,''),  nullif(s.rad_obs2,'')),  ''), s.rad_status),
      ('limpeza_tanque',      s.tanq_realizada,  s.tanq_vencimento,
         nullif(concat_ws(' | ', nullif(s.tanq_obs,''), nullif(s.tanq_obs2,'')), ''), s.tanq_status)
    ) as v(item_codigo, dt_real, dt_venc, obs, st)
    where s.import_id = p_import_id
  ),
  parsed as (
    select
      u.import_id, u.linha_planilha, u.numero, u.cliente, u.vendedor,
      (trim(coalesce(u.ativo,'')) ilike 'sim%') as contrato_ativo,
      u.item_codigo,
      public.parse_data_pred(u.dt_real) as d_real,
      public.parse_data_pred(u.dt_venc) as d_venc,
      nullif(trim(coalesce(u.obs,'')), '') as observacao,
      nullif(trim(coalesce(u.st,'')),  '') as status_planilha,
      (public.eh_nao_aplica(u.dt_real) or public.eh_nao_aplica(u.dt_venc)
       or public.eh_nao_aplica(u.st))      as nao_aplica,
      i.periodicidade_meses,
      to_jsonb(u) - 'import_id'            as origem
    from unpiv u
    join public.preditiva_itens i on i.codigo = u.item_codigo
  ),
  calc as (
    select p.*,
      coalesce(p.d_venc, case when p.d_real is not null and p.periodicidade_meses is not null
        then (p.d_real + (p.periodicidade_meses || ' months')::interval)::date end) as venc_final,
      coalesce(p.d_real, case when p.d_venc is not null and p.periodicidade_meses is not null
        then (p.d_venc - (p.periodicidade_meses || ' months')::interval)::date end) as real_final
    from parsed p
  )
  insert into public.preditivas_atg (
    import_id, linha_planilha, numero_planilha, cliente_planilha, vendedor,
    contrato_ativo, gerador_id, omie_cliente_id, cnpj_limpo,
    match_score, match_status, match_candidatos,
    item_codigo, data_realizada, data_vencimento,
    sem_data, nao_aplica, observacao, status_planilha, origem_planilha
  )
  select
    c.import_id, c.linha_planilha, c.numero, c.cliente, c.vendedor,
    c.contrato_ativo,
    mt.gerador_id, mt.omie_cliente_id, mt.cnpj_limpo,
    mt.score, mt.match_status, mt.candidatos,
    c.item_codigo,
    case when c.nao_aplica then null else c.real_final end,
    case when c.nao_aplica then null else c.venc_final end,
    (not c.nao_aplica and c.venc_final is null),
    c.nao_aplica,
    c.observacao, c.status_planilha, c.origem
  from calc c
  join _mt mt on mt.cliente = c.cliente;

  -- ---------------- OXICATALISADOR (cadastro) -------------------------
  insert into public.oxicatalisador_atg (
    import_id, linha_planilha, cliente_planilha,
    gerador_id, cnpj_limpo, match_score, match_status,
    possui, situacao, bitola, bitola_pendente
  )
  select
    s.import_id, s.linha_planilha, s.cliente,
    mt.gerador_id, mt.cnpj_limpo, mt.score, mt.match_status,
    case
      when trim(coalesce(s.oxi_tem,'')) ilike 'sim%' then true
      when trim(coalesce(s.oxi_tem,'')) ilike 'n_o%' then false
      else null
    end,
    case
      when trim(coalesce(s.oxi_tem,'')) ilike 'sim%'      then 'sim'
      when trim(coalesce(s.oxi_tem,'')) ilike 'n_o%'      then 'nao'
      when trim(coalesce(s.oxi_tem,'')) ilike 'pend%'     then 'pendencia'
      when trim(coalesce(s.oxi_tem,'')) ilike 'sem info%' then 'sem_informacao'
      else 'sem_informacao'
    end,
    nullif(case when trim(coalesce(s.oxi_bitola,'')) ilike 'pend%'
                then null else trim(s.oxi_bitola) end, ''),
    (trim(coalesce(s.oxi_bitola,'')) ilike 'pend%')
  from public.preditivas_staging s
  join _mt mt on mt.cliente = s.cliente
  where s.import_id = p_import_id and s.cliente is not null;

  return query
  select count(*)::int,
         count(*) filter (where p.match_status = 'auto')::int,
         count(*) filter (where p.match_status = 'revisar')::int,
         count(*) filter (where p.match_status = 'sem_match')::int
  from public.preditivas_atg p where p.import_id = p_import_id;
end $function$

score_nomes

Regras

  • Nota = 0,30 × cobertura do peso das palavras + 0,45 × se a palavra mais rara do nome do banco aparece + 0,25 × parecido (palavras em comum sobre o total com penalidades).
  • Palavras raras pesam mais (peso vem da mv_idf_tokens).

Código da função

CREATE OR REPLACE FUNCTION public.score_nomes(p_planilha text, p_banco text)
 RETURNS numeric
 LANGUAGE plpgsql
 STABLE
 SET search_path TO 'public', 'extensions', 'pg_catalog'
AS $function$
declare
  tp text[] := public.tokens_nome(p_planilha);
  tb text[] := public.tokens_nome(p_banco);
  inter    text[];
  sobra_b  text[];
  sobra_p  text[];
  w_inter numeric; w_b numeric;
  cov_b numeric; cov_raro numeric; jac numeric;
  max_idf numeric; tok_raro text;
  pen_sb numeric; pen_sp numeric; denom numeric;
  n_inter int;
begin
  if tp is null or tb is null
     or array_length(tp,1) is null or array_length(tb,1) is null then
    return 0;
  end if;

  select array_agg(t) into inter   from unnest(tb) t where t = any(tp);
  if inter is null then return 0; end if;

  select array_agg(t) into sobra_b from unnest(tb) t where not (t = any(tp));
  select array_agg(t) into sobra_p from unnest(tp) t where not (t = any(tb));
  sobra_b := coalesce(sobra_b, '{}');
  sobra_p := coalesce(sobra_p, '{}');

  w_inter := public.peso_tokens(inter);
  w_b     := public.peso_tokens(tb);
  if w_b = 0 then return 0; end if;
  cov_b := w_inter / w_b;

  -- token mais raro (maior idf) do nome do banco
  select t, public.idf(t) into tok_raro, max_idf
  from unnest(tb) t order by public.idf(t) desc, t asc limit 1;
  if max_idf is null or max_idf = 0 then return 0; end if;

  cov_raro := case when tok_raro = any(inter) then 1.0 else 0.0 end;

  -- penalidades quadraticas, relativas ao token mais raro do banco.
  -- token do banco com idf proximo do maximo (ex 'LCB') pune quase 1.0;
  -- token acessorio (ex 'ARTES') pune pouco.
  select coalesce(sum(power(public.idf(t) / max_idf, 2)), 0) into pen_sb
  from unnest(sobra_b) t;

  select coalesce(sum(power(public.idf(t) / max_idf, 2)), 0)
         * (case when cov_raro = 1.0 then 0.05 else 0.30 end)
    into pen_sp
  from unnest(sobra_p) t;

  n_inter := array_length(inter, 1);
  denom   := n_inter + pen_sb + pen_sp;
  jac     := case when denom > 0 then n_inter::numeric / denom else 0 end;

  return round((0.30 * cov_b + 0.45 * cov_raro + 0.25 * jac)::numeric, 4);
end $function$

CREATE OR REPLACE FUNCTION public.validar_cadastro(p_entidade text, p_id bigint, p_usuario text, p_valor boolean DEFAULT true)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
begin
  if p_entidade = 'cliente' then
    update public.omie_clientes
      set validado = p_valor,
          validado_por = case when p_valor then p_usuario else null end,
          validado_em  = case when p_valor then now() else null end
    where id = p_id;
  elsif p_entidade = 'gerador' then
    update public.geradores_atg
      set validado = p_valor,
          validado_por = case when p_valor then p_usuario else null end,
          validado_em  = case when p_valor then now() else null end
    where id = p_id;
  else
    raise exception 'entidade invalida: % (use cliente ou gerador)', p_entidade;
  end if;

  if not found then
    raise exception '% % nao encontrado', p_entidade, p_id;
  end if;

  insert into public.log_edicoes (entidade, entidade_id, acao, usuario)
  values (p_entidade, p_id, case when p_valor then 'validou' else 'desvalidou' end, p_usuario);
end $function$
